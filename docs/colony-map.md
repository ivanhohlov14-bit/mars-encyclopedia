<h2 class="ca-h2 ca-observe">❓ Частые вопросы</h2>
<div class="ca-faq ca-observe">
  <details class="ca-faq-item">
    <summary>
      <span>Почему нельзя колонизировать Луну, если она ближе?</span>
      <span class="ca-faq-icon">▸</span>
    </summary>
    <div class="ca-faq-answer">
      Луна ближе (3 дня полёта), но на ней <strong>нет воды и атмосферы</strong>.
      Ночью температура падает до −173 °C, а радиация ещё выше, чем на Марсе.
      Марс имеет воду, слабую атмосферу и больше ресурсов для долгосрочного
      выживания.
    </div>
  </details>

  <details class="ca-faq-item">
    <summary>
      <span>Сколько стоит полёт на Марс?</span>
      <span class="ca-faq-icon">▸</span>
    </summary>
    <div class="ca-faq-answer">
      Оценки разнятся. Миссия NASA по программе Artemis обойдётся в
      <strong>~100 млрд долларов</strong>. Илон Маск обещает снизить стоимость
      одного запуска Starship до <strong>2–10 млн долларов</strong> при массовом
      производстве.
    </div>
  </details>

  <details class="ca-faq-item">
    <summary>
      <span>Как защититься от радиации на поверхности?</span>
      <span class="ca-faq-icon">▸</span>
    </summary>
    <div class="ca-faq-answer">
      Три варианта: <strong>толща реголита</strong> (1–2 метра над головой),
      <strong>вода</strong> как щит (отлично поглощает излучение),
      <strong>подземные убежища</strong> в лавовых трубках.
      На поверхности — только короткие выходы в защищённом снаряжении.
    </div>
  </details>

  <details class="ca-faq-item">
    <summary>
      <span>Можно ли дышать марсианским воздухом?</span>
      <span class="ca-faq-icon">▸</span>
    </summary>
    <div class="ca-faq-answer">
      Нет. Атмосфера Марса — <strong>95% CO₂</strong>, давление в 170 раз ниже
      земного. Человек задохнётся и получит кессонную болезнь за секунды.
      Воздух для базы придётся производить изо льда (электролиз воды).
    </div>
  </details>

  <details class="ca-faq-item">
    <summary>
      <span>Что случится с человеком при разгерметизации?</span>
      <span class="ca-faq-icon">▸</span>
    </summary>
    <div class="ca-faq-answer">
      Миф про «мгновенное закипание» — преувеличение. Человек потеряет
      сознание за <strong>10–15 секунд</strong> из-за гипоксии.
      Кровь не закипит — спасёт эластичность сосудов. Смерть наступит
      через <strong>1–2 минуты</strong> без скафандра.
    </div>
  </details>

  <details class="ca-faq-item">
    <summary>
      <span>Сколько воды нужно везти с Земли?</span>
      <span class="ca-faq-icon">▸</span>
    </summary>
    <div class="ca-faq-answer">
      Изначально — <strong>нисколько</strong>, если выбрать Amazonis Planitia.
      Вода есть прямо под ногами. На первое время достаточно
      привезти систему переработки — она замкнёт цикл на 90%.
    </div>
  </details>

  <details class="ca-faq-item">
    <summary>
      <span>Как долго продлится полёт?</span>
      <span class="ca-faq-icon">▸</span>
    </summary>
    <div class="ca-faq-answer">
      <strong>7–9 месяцев</strong> в одну сторону. Расстояние между планетами
      меняется от <strong>54.6 млн км</strong> до <strong>401 млн км</strong>.
      Стартовое окно открывается раз в <strong>26 месяцев</strong>.
    </div>
  </details>

  <details class="ca-faq-item">
    <summary>
      <span>Что делать с отходами жизнедеятельности?</span>
      <span class="ca-faq-icon">▸</span>
    </summary>
    <div class="ca-faq-answer">
      Замкнутый цикл: <strong>переработка воды</strong> (как на МКС),
      <strong>компостирование</strong> биомассы, <strong>использование</strong>
      в качестве удобрений для теплиц. Минимум отходов — максимум пользы.
    </div>
  </details>
</div>

/* ═══ SCROLL OBSERVE ═══ */
function initObserve(){
  var items = document.querySelectorAll('.ca-observe');
  if (!items.length) return;
  if (!('IntersectionObserver' in window)){
    items.forEach(function(el){ el.classList.add('ca-visible'); });
    return;
  }
  var io = new IntersectionObserver(function(entries){
    entries.forEach(function(e){
      if (e.isIntersecting){
        e.target.classList.add('ca-visible');
        io.unobserve(e.target);
      }
    });
  }, { threshold: 0.12, rootMargin: '0px 0px -40px 0px' });
  items.forEach(function(el){ io.observe(el); });
}

/* ═══ СЧЁТЧИКИ ═══ */
function animateCounter(el){
  var target = parseFloat(el.dataset.target || '0');
  var decimals = parseInt(el.dataset.decimals || '0', 10);
  var duration = 1600;
  var start = performance.now();
  function tick(now){
    var p = Math.min((now - start) / duration, 1);
    var eased = 1 - Math.pow(1 - p, 3);
    var val = target * eased;
    el.textContent = decimals > 0 ? val.toFixed(decimals) : Math.round(val).toString();
    if (p < 1) requestAnimationFrame(tick);
    else el.textContent = decimals > 0 ? target.toFixed(decimals) : target.toString();
  }
  requestAnimationFrame(tick);
}

function initCounters(){
  var counters = document.querySelectorAll('.ca-counter');
  if (!counters.length) return;
  if (!('IntersectionObserver' in window)){
    counters.forEach(animateCounter);
    return;
  }
  var io = new IntersectionObserver(function(entries){
    entries.forEach(function(e){
      if (e.isIntersecting){
        animateCounter(e.target);
        io.unobserve(e.target);
      }
    });
  }, { threshold: 0.5 });
  counters.forEach(function(el){ io.observe(el); });
}

      if (toggle){
        toggle.textContent = isCollapsed ? '+' : '−';
        toggle.setAttribute('aria-label', isCollapsed ? 'Развернуть' : 'Свернуть');
      }
    }
    if (head) head.addEventListener('click', doToggle);
    if (toggle) toggle.addEventListener('click', function(e){
      e.stopPropagation();
      doToggle(e);
    });
  });
}

/* ═══ КАЛЬКУЛЯТОР ВОДЫ ═══ */
function initCalculator(){
  var slider = document.getElementById('ca-colonists');
  if (!slider) return;

  var valueEl = document.getElementById('ca-colonists-value');
  var drinkEl = document.getElementById('ca-water-drink');
  var totalEl = document.getElementById('ca-water-total');
  var iceEl = document.getElementById('ca-water-ice');
  var earthEl = document.getElementById('ca-water-earth');

  // Нормы NASA (л/чел/день):
  // питьё = 3, еда = 3, гигиена = 20 → итого ~26 л
  var DRINK_PER_PERSON = 3;
  var TOTAL_PER_PERSON = 26;   // без рециркуляции
  var ICE_PER_LITER = 0.9;     // 1 л воды ≈ 0.9 кг льда (плотность <1)
  var DAYS_PER_YEAR = 365;

  function fmt(n, decimals){
    decimals = decimals || 0;
    var s = n.toFixed(decimals);
    // Разделитель тысяч пробелом
    var parts = s.split('.');
    parts[0] = parts[0].replace(/\B(?=(\d{3})+(?!\d))/g, ' ');
    return parts.join('.');
  }

  function update(){
    var n = parseInt(slider.value, 10) || 0;
    valueEl.textContent = n;

    var drink = n * DRINK_PER_PERSON;
    var total = n * TOTAL_PER_PERSON;
    var iceTons = (total * ICE_PER_LITER) / 1000;   // тонн в сутки
    var earthTons = iceTons * DAYS_PER_YEAR;         // тонн в год

    drinkEl.textContent = fmt(drink);
    totalEl.textContent = fmt(total);
    iceEl.textContent = fmt(iceTons, iceTons < 10 ? 1 : 0);
    earthEl.textContent = fmt(earthTons, 0);
  }

  slider.addEventListener('input', update);
  update();
}

/* ═══ АНИМАЦИЯ ПРОГРЕСС-БАРОВ ПРИ ПОЯВЛЕНИИ ═══ */
function initBars(){
  var bars = document.querySelectorAll('.ca-bar-fill, .ca-compare-fill, .ca-criterion-bar-fill');
  if (!bars.length) return;
  if (!('IntersectionObserver' in window)) return;

  bars.forEach(function(bar){
    var targetWidth = getComputedStyle(bar).getPropertyValue('--fill').trim() || '100%';
    bar.dataset.targetWidth = targetWidth;
    bar.style.width = '0';
  });

  var io = new IntersectionObserver(function(entries){
    entries.forEach(function(e){
      if (e.isIntersecting){
        var bar = e.target;
        var w = bar.dataset.targetWidth;
        bar.style.width = w;
        io.unobserve(bar);
      }
    });
  }, { threshold: 0.3 });

  bars.forEach(function(bar){ io.observe(bar); });
}
