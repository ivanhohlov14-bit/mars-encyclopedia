<!-- INTRO -->
<div class="rv-intro rv-observe">
  <p>
    4 июля 1997 года на Марсе сел первый ровер — <strong>Sojourner</strong>.
    Он был размером с микроволновку и весил 10 кг. Никто не думал,
    что он проработает <strong>83 дня</strong> вместо запланированных 7.
  </p>
  <p>
    С тех пор NASA отправило ещё четыре марсохода. Каждый был умнее
    предыдущего. Каждый делал открытия, которые меняли учебники.
    Вот их полная история.
  </p>
</div>

<h2 class="rv-h2 rv-observe">❓ Частые вопросы</h2>
<div class="rv-faq rv-observe">
  <details class="rv-faq-item">
    <summary><span>Сколько всего марсоходов было на Марсе?</span><span class="rv-faq-icon">▸</span></summary>
    <div class="rv-faq-answer">
      <strong>5 от NASA</strong>: Sojourner (1997), Spirit и Opportunity (2004),
      Curiosity (2012), Perseverance (2021). Плюс китайский <strong>Чжучжун</strong>
      (2021) — но он уже завершил работу.
    </div>
  </details>

  <details class="rv-faq-item">
    <summary><span>Какой ровер самый успешный?</span><span class="rv-faq-icon">▸</span></summary>
    <div class="rv-faq-answer">
      <strong>Opportunity</strong> — по долговечности (15 лет вместо 90 дней)
      и по пробегу (45 км — мировой рекорд). <strong>Curiosity</strong> — по
      научным открытиям. <strong>Perseverance</strong> — по технологиям
      (вертолёт, сбор образцов).
    </div>
  </details>

  <details class="rv-faq-item">
    <summary><span>Почему Opportunity погиб?</span><span class="rv-faq-icon">▸</span></summary>
    <div class="rv-faq-answer">
      В июне 2018 года Марс накрыла <strong>глобальная пылевая буря</strong>.
      Небо потемнело на месяцы, солнечные панели перестали
      вырабатывать энергию. Opportunity перешёл в спящий режим
      и не проснулся. NASA пыталось связаться <strong>8 месяцев</strong> —
      безуспешно.
    </div>
  </details>

  <details class="rv-faq-item">
    <summary><span>Что делает Perseverance сейчас?</span><span class="rv-faq-icon">▸</span></summary>
    <div class="rv-faq-answer">
      Исследует кратер Езеро — древнюю речную дельту.
      Собирает <strong>образцы пород</strong> в специальные
      капсулы. Их заберёт будущая миссия <strong>Mars Sample
      Return</strong> и вернёт на Землю. Ожидается в 2030-х.
    </div>
  </details>

  <details class="rv-faq-item">
    <summary><span>Сколько стоит один марсоход?</span><span class="rv-faq-icon">▸</span></summary>

      (Perseverance). Curiosity стоил <strong>$2.5 млрд</strong>.
      Основные затраты — не сам робот, а запуск, связь и
      команда управления на 10+ лет.
    </div>
  </details>

  <details class="rv-faq-item">
    <summary><span>Можно ли управлять марсоходом в реальном времени?</span><span class="rv-faq-icon">▸</span></summary>

      <strong>автономно</strong>: получают команду, выполняют,

      игра.
    </div>

/* ═══ SCROLL OBSERVE ═══ */
function initObserve(){
  var items = document.querySelectorAll('.rv-observe');
  if (!items.length) return;
  if (!('IntersectionObserver' in window)){
    items.forEach(function(el){ el.classList.add('rv-visible'); });
    return;
  }
  var io = new IntersectionObserver(function(entries){
    entries.forEach(function(e){
      if (e.isIntersecting){
        e.target.classList.add('rv-visible');
        io.unobserve(e.target);
      }
    });
  }, { threshold: 0.1, rootMargin: '0px 0px -40px 0px' });
  items.forEach(function(el){ io.observe(el); });
}

/* ═══ СЧЁТЧИКИ ═══ */
function animateCounter(el){
  var target = parseFloat(el.dataset.target || '0');
  var format = el.dataset.format;
  var duration = 1800;
  var start = performance.now();
  function fmt(n){
    if (format === 'million') return (n/1000000).toFixed(1) + ' млн';
    return Math.round(n).toString();
  }
  function tick(now){
    var p = Math.min((now - start) / duration, 1);
    var eased = 1 - Math.pow(1 - p, 3);
    var val = target * eased;
    el.textContent = fmt(val);
    if (p < 1) requestAnimationFrame(tick);
    else el.textContent = fmt(target);
  }
  requestAnimationFrame(tick);
}
function initCounters(){
  var counters = document.querySelectorAll('.rv-counter');
  if (!counters.length) return;
  if (!('IntersectionObserver' in window)){
    counters.forEach(animateCounter);
    return;
  }
  var io = new IntersectionObserver(function(entries){
    entries.forEach(function(e){
      if (e.isIntersecting){
        animateCounter(e.target);
        io.unobserve(e.target);
      }
    });
  }, { threshold: 0.5 });
  counters.forEach(function(el){ io.observe(el); });
}
