<h2 class="ml-h2 ml-observe">❓ Частые вопросы</h2>
<div class="ml-faq ml-observe">
  <details class="ml-faq-item">
    <summary><span>Кто платит за билет на Марс?</span><span class="ml-faq-icon">▸</span></summary>
    <div class="ml-faq-answer">
      Пока — частные компании (SpaceX) и государства. Первый билет
      стоит около <strong>$500 000</strong>. В будущем — колонисты
      работают «за долю» в колонии, как инвесторы.
    </div>
  </details>

  <details class="ml-faq-item">
    <summary><span>Можно ли улететь обратно?</span><span class="ml-faq-icon">▸</span></summary>
    <div class="ml-faq-answer">
      Технически да, но окно для возврата — раз в 26 месяцев (когда
      Земля и Марс на минимальном расстоянии). Первые колонисты,
      скорее всего, <strong>никогда не вернутся</strong> — слишком дорого.
    </div>
  </details>

  <details class="ml-faq-item">
    <summary><span>Что с деньгами, заработанными на Земле?</span><span class="ml-faq-icon">▸</span></summary>
    <div class="ml-faq-answer">
      Тратить негде. В колонии появятся свои деньги или система
      обмена. Земные рубли/доллары останутся как <strong>квази-историческая
      единица</strong> — на них можно купить только обратный билет.
    </div>
  </details>

  <details class="ml-faq-item">
    <summary><span>Можно ли убить человека на Марсе?</span><span class="ml-faq-icon">▸</span></summary>
    <div class="ml-faq-answer">
      Юридически — убийство это убийство. Но реально: <strong>колония
      будет решать сама</strong>. Возможные наказания — изоляция, изгнание
      (смерть), принудительные работы. <strong>Казни — вероятны</strong>.
    </div>
  </details>

  <details class="ml-faq-item">
    <summary><span>Что если колония объявит независимость?</span><span class="ml-faq-icon">▸</span></summary>
    <div class="ml-faq-answer">
      По международному праву — <strong>проблема</strong>. Договор о космосе
      (1967) запрещает претензии на территории. Но если колония
      живёт автономно 20 лет, Земля <strong>не сможет вернуть контроль</strong>.
    </div>
  </details>

  <details class="ml-faq-item">
    <summary><span>Как решать конфликты между колонистами?</span><span class="ml-faq-icon">▸</span></summary>
    <div class="ml-faq-answer">
      В маленькой колонии работает <strong>медиация</strong> — старейшины
      или психолог. В большой — суд. Но главное: <strong>нельзя
      поссориться и уйти</strong> — везде 50 м² общего пространства.
    </div>
  </details>
</div>

  stability: 50,
  efficiency: 50,
  choices: {}
};

function renderChoices(){
  var wrap = document.getElementById('ml-choices');
  wrap.innerHTML = CHOICES.map(function(c){
    return '<div class="ml-sim-group">' +
      '<div class="ml-sim-group-title">' + c.title + '</div>' +
      '<div class="ml-sim-options">' +
        c.options.map(function(o){
          return '<button type="button" class="ml-sim-opt" data-group="' + c.id + '" data-opt="' + o.id + '">' + o.label + '</button>';
        }).join('') +
      '</div>' +
    '</div>';
  }).join('');

  wrap.querySelectorAll('.ml-sim-opt').forEach(function(btn){
    btn.addEventListener('click', function(){
      var group = btn.dataset.group;
      var optId = btn.dataset.opt;

      // снимаем актив со всех в этой группе
      wrap.querySelectorAll('.ml-sim-opt[data-group="' + group + '"]').forEach(function(b){
        b.classList.remove('active');
      });
      btn.classList.add('active');

      // обновляем выбор
      state.choices[group] = optId;
      recalc();
    });
  });
}

function recalc(){
  // сброс
  var f = 50, s = 50, e = 50;

  CHOICES.forEach(function(c){
    var chosen = state.choices[c.id];
    if (!chosen) return;
    var opt = c.options.filter(function(o){ return o.id === chosen; })[0];
    if (!opt) return;
    f += opt.freedom;
    s += opt.stability;
    e += opt.efficiency;
  });

  state.freedom = Math.max(0, Math.min(100, f));
  state.stability = Math.max(0, Math.min(100, s));
  state.efficiency = Math.max(0, Math.min(100, e));

  document.getElementById('ml-freedom').textContent = state.freedom;
  document.getElementById('ml-stability').textContent = state.stability;
  document.getElementById('ml-efficiency').textContent = state.efficiency;
  document.getElementById('ml-freedom-bar').style.width = state.freedom + '%';
  document.getElementById('ml-stability-bar').style.width = state.stability + '%';
  document.getElementById('ml-efficiency-bar').style.width = state.efficiency + '%';

  updateVerdict();
}

/* ═══ OBSERVE ═══ */
function initObserve(){
  var items = document.querySelectorAll('.ml-observe');
  if (!items.length) return;
  if (!('IntersectionObserver' in window)){
    items.forEach(function(el){ el.classList.add('ml-visible'); });
    return;
  }
  var io = new IntersectionObserver(function(entries){
    entries.forEach(function(e){
      if (e.isIntersecting){
        e.target.classList.add('ml-visible');
        io.unobserve(e.target);
      }
    });
  }, { threshold: 0.1, rootMargin: '0px 0px -40px 0px' });
  items.forEach(function(el){ io.observe(el); });
}

/* ═══ COUNTERS ═══ */
function animateCounter(el){
  var target = parseFloat(el.dataset.target || '0');
  var duration = 1800;
  var start = performance.now();
  function tick(now){
    var p = Math.min((now - start) / duration, 1);
    var eased = 1 - Math.pow(1 - p, 3);
    el.textContent = Math.round(target * eased).toString();
    if (p < 1) requestAnimationFrame(tick);
    else el.textContent = target.toString();
  }
  requestAnimationFrame(tick);
}

function initCounters(){
  var counters = document.querySelectorAll('.ml-counter');
  if (!counters.length) return;
  if (!('IntersectionObserver' in window)){
    counters.forEach(animateCounter);
    return;
  }
  var io = new IntersectionObserver(function(entries){
    entries.forEach(function(e){
      if (e.isIntersecting){
        animateCounter(e.target);
        io.unobserve(e.target);
      }
    });
  }, { threshold: 0.5 });
  counters.forEach(function(el){ io.observe(el); });
}
